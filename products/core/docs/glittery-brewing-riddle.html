Migration Plan: Nessie+Iceberg ? DuckLake+DuckDB

Context

VaultDB's current data layer uses Nessie+Iceberg which introduces excessive operational complexity: a running Nessie Java server, custom REST API bridge code, Iceberg metadata overhead for small writes, and S3/MinIO required even for local dev. The sandbox pattern (per-user branch with small delta writes) is poorly served by Iceberg's heavy snapshot model.

DuckLake eliminates all of this: metadata in a single DuckDB file, data inlining for small writes, native SQL-level merge, zero-config local dev, and zero-copy Arrow interop. The DuckLake metadata catalog file is downloaded from S3 after authentication, and DuckDB secrets are created from auth credentials for S3 data access.

Sandbox isolation: Schema-per-user within a single DuckLake catalog (no branching needed). Metadata storage: DuckDB .ducklake file on S3, downloaded after auth via existing download_file.

1.4 Table access — replaces get_table_at_ref()

def scan_table(
    self,
    table_name: str,
    schema: str = MAIN_SCHEMA,
    row_filter: str | None = None,
    params: list | None = None,
) -> pa.Table:
    """Scan table, return Arrow. Replaces get_table_at_ref() + .scan().to_arrow()."""
    query = f"SELECT * FROM {self.catalog_name}.{schema}.{table_name}"
    if row_filter:
        query += f" WHERE {row_filter}"
    return self.con.execute(query, params or []).fetch_arrow_table()

def scan_table_at_version(
    self,
    table_name: str,
    version: int | None = None,
    as_of: datetime | None = None,
    schema: str = MAIN_SCHEMA,
    row_filter: str | None = None,
    params: list | None = None,
) -> pa.Table:
    """Time-travel query. Replaces get_table_at_snapshot()."""
    tbl = f"{self.catalog_name}.{schema}.{table_name}"
    if version is not None:
        tbl += f" AT (VERSION => {version})"
    elif as_of is not None:
        tbl += f" AT (TIMESTAMP => '{as_of.isoformat()}')"
    query = f"SELECT * FROM {tbl}"
    if row_filter:
        query += f" WHERE {row_filter}"
    return self.con.execute(query, params or []).fetch_arrow_table()

class Reference(Column):
    """Foreign key reference to another Model. Stored as VARCHAR (the referenced model's PK value)."""
    def __init__(self, model: type, **kwargs):
        super().__init__(**kwargs)
        self.referenced_model = model  # the target Model class

    @property
    def db_type(self) -> str:
        return "VARCHAR"  # FK value serialized as string

    @property
    def fk_constraint(self) -> str:
        """Returns SQL REFERENCES clause for DDL."""
        tbl = self.referenced_model.__table_name__
        pk = self.referenced_model.__primary_key__
        return f"REFERENCES main.{tbl}({pk})"

Serialization helpers (add to columns.py or a serialization.py utility):

import json

def serialize_column_value(col: Column, value) -> str | None:
    """Serialize container/object types to JSON string for DuckDB storage."""
    if value is None:
        return None
    if isinstance(col, (Map, Set, Dict, List, ObjectType)):
        if isinstance(value, set):
            return json.dumps(list(value))
        return json.dumps(value if not hasattr(value, '__dict__') else value.__dict__)
    return value

def deserialize_column_value(col: Column, raw: str | None):
    """Deserialize JSON string back to Python type."""
    if raw is None:
        return None
    if isinstance(col, (Map, Dict)):
        return json.loads(raw)
    if isinstance(col, Set):
        return set(json.loads(raw))
    if isinstance(col, List):
        return json.loads(raw)
    if isinstance(col, ObjectType):
        return col.usertype(**json.loads(raw))
    return raw

Phase 5: Model sync_model() — Replace Iceberg Schema with DuckDB DDL

File: src/vaultdb/data/model.py

5.1 Replace sync_model() (~lines 264-362)

DuckDB natively supports PRIMARY KEY — use it directly in DDL instead of managing uniqueness ourselves. Also emit REFERENCES constraints for Reference columns (foreign keys).

@classmethod
def sync_model(cls) -> None:
    """Sync model schema with DuckLake catalog via DuckDB DDL."""
    catalog = cls.__catalog__

    col_defs = []
    fk_constraints = []

    for name, col in cls._columns.items():
        if not col.persistent:
            continue
        line = f"{name} {col.db_type}"
        if name == cls.__primary_key__:
            line += " PRIMARY KEY"
        col_defs.append(line)
        # Collect FK constraints for Reference columns
        if isinstance(col, Reference):
            fk_constraints.append(
                f"FOREIGN KEY ({name}) {col.fk_constraint}"
            )

    col_defs.append("_is_deleted BOOLEAN DEFAULT FALSE")
    all_defs = col_defs + fk_constraints

    ddl = f"""
        CREATE TABLE IF NOT EXISTS {catalog.catalog_name}.main.{cls.__table_name__} (
            {', '.join(all_defs)}
        )
    """
    catalog.con.execute(ddl)

    # Schema evolution: ADD COLUMN for missing columns
    existing = set(catalog.con.execute(
        f"SELECT column_name FROM information_schema.columns "
        f"WHERE table_schema = 'main' AND table_name = '{cls.__table_name__}'"
    ).fetchall())
    # ... ALTER TABLE ADD COLUMN for any missing

Note: cls.__primary_key__ must be declared on the Model class (e.g. __primary_key__ = "id"). Remove any manual PK uniqueness enforcement from save() or mutator.py — DuckDB enforces it at the DB level.

def upsert(self, table_name: str, df: pd.DataFrame, primary_key: str, schema: str = MAIN_SCHEMA) -> None:
    """INSERT OR REPLACE using DuckDB's native primary key enforcement."""
    tbl = f"{self.catalog_name}.{schema}.{table_name}"
    # Register df as a view then INSERT OR REPLACE
    self.con.register("_upsert_df", df)

Phase 7: UserType — Drop PyIceberg StructType; Require UserType for Container Columns

File: src/vaultdb/data/usertype.py

Replace pyiceberg.types.StructType / NestedField with a plain Python base class. UserTypes serve as the schema definition for container column contents and are also saved as their own DuckDB table (so the schema is queryable and typed).

class UserType:
    """Base class for structured types used in Map/Set/List/ObjectType columns."""
    _columns: dict[str, Column] = {}  # declared on subclass

    @classmethod
    def get_struct_ddl(cls) -> str:
        """Return DuckDB STRUCT type string for schema documentation."""
        fields = [f"{name} {col.db_type}" for name, col in cls._columns.items()]
        return f"STRUCT({', '.join(fields)})"

    @classmethod
    def sync_usertype(cls, catalog) -> None:
        """
        Persist the UserType schema as a metadata table in DuckDB.
        Table name: _usertype_<classname> in main schema.
        This allows introspection and future typed MAP/STRUCT support.
        """
        tbl = f"_usertype_{cls.__name__.lower()}"
        col_defs = [f"{name} {col.db_type}" for name, col in cls._columns.items()]
        ddl = f"""
            CREATE TABLE IF NOT EXISTS {catalog.catalog_name}.main.{tbl} (
                {', '.join(col_defs)}
            )
        """
        catalog.con.execute(ddl)

Rule: Any Model column using Map, Set, Dict, List, or ObjectType must declare a UserType subclass describing the contained schema. sync_model() must call col.usertype.sync_usertype(catalog) for each such column so the schema is registered in DuckDB.
